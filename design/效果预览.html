// CSS - 悬浮球基础样式
.float-ball {
    position: absolute;
    right: 20px;
    bottom: 20px;
    width: 16px;
    height: 16px;
    border-radius: 50%;
    background: rgba(255, 255, 255, 0.3);
    transition: all 0.3s ease;
}

// JavaScript - 状态管理
function setState(state) {
    const ball = document.getElementById('floatBall');
    const window = document.getElementById('floatWindow');
    
    // 移除所有状态类
    ball.className = 'float-ball';
    window.className = 'float-window';
    
    // 添加新状态
    ball.classList.add(state);
    if (state !== 'idle') {
        window.classList.add(state, 'active');
    }
    
    // 更新内容
    updateContent(state);
    
    // 自动收起（成功/错误状态）
    if (state === 'success' || state === 'error') {
        setTimeout(resetToIdle, 2000);
    }
}
